// 🔧 FUNÇÃO: LIBERAR ACESSO (API LOCAL + RADIUS)
// ===============================
async function liberarAcessoDual(mac, pop_ip = '192.168.32.1') {
    let viaAPI = false;
    let viaRADIUS = false;
    let erros = [];
    
    // TENTATIVA 1: API Direta no MikroTik
    try {
        const conn = new RouterOSAPI({
            host: pop_ip,
            user: 'ms_api',
            password: 'REDACTED',
            port: 8728,
            timeout: 5
        });
        
        await conn.connect();
        await conn.write('/ip/hotspot/user/add', [
            `=name=${mac}`,
            `=password=${mac}`,
            '=profile=default'
        ]);
        await conn.close();
        
        viaAPI = true;
        console.log(`✅ Usuário ${mac} liberado via API local (MikroTik)`);
    } catch (error) {
        erros.push(`API: ${error.message}`);
        console.log(`⚠️ API falhou para ${mac}: ${error.message}`);
    }
    
    // TENTATIVA 2: RADIUS (Supabase)
    try {
        const dataExpiracao = new Date();
        dataExpiracao.setMinutes(dataExpiracao.getMinutes() + 15);
        
        const { data: existing } = await supabase
            .from('radreply')
            .select('id')
            .eq('username', mac)
            .single();
        
        if (existing) {
            await supabase
                .from('radreply')
                .update({
                    status: 'active',
                    expira_em: dataExpiracao.toISOString(),
                    updated_at: new Date().toISOString()
                })
                .eq('username', mac);
        } else {
            await supabase
                .from('radreply')
                .insert({
                    username: mac,
                    attribute: 'Cleartext-Password',
                    op: ':=',
                    value: mac,
                    plano: 'teste_gratis',
                    status: 'active',
                    expira_em: dataExpiracao.toISOString(),
                    created_at: new Date().toISOString()
                });
        }
        
        viaRADIUS = true;
        console.log(`✅ Usuário ${mac} liberado via RADIUS (Supabase)`);
    } catch (error) {
        erros.push(`RADIUS: ${error.message}`);
        console.log(`⚠️ RADIUS falhou para ${mac}: ${error.message}`);
    }
    
    return {
        success: viaAPI || viaRADIUS,
        viaAPI,
        viaRADIUS,
        erros
    };
}

const verificarToken = (req, res, next) => {
    const token = req.headers.authorization?.split(' ')[1];
    if (!token) {
        return res.status(401).json({ error: 'Token não fornecido' });
    }
    try {
        const decoded = jwt.verify(token, JWT_SECRET);
        req.user = decoded;
        next();
    } catch (error) {
        return res.status(401).json({ error: 'Token inválido' });
    }
};

app.post('/api/login', async (req, res) => {
    try {
        const { username, password } = req.body;

        if (!username || !password) {
            return res.status(400).json({ error: 'Usuário e senha obrigatórios' });
        }

        const { data: admin, error } = await supabase
            .from('admins')
            .select('*')
            .eq('username', username)
            .single();

        if (error || !admin) {
            return res.status(401).json({ error: 'Usuário ou senha incorretos' });
        }

        if (admin.password !== password) {
            return res.status(401).json({ error: 'Usuário ou senha incorretos' });
        }

        const token = jwt.sign(
            { id: admin.id, username: admin.username, role: admin.role },
            JWT_SECRET,
            { expiresIn: '24h' }
        );

        console.log(`✅ Login bem-sucedido: ${username}`);
        return res.json({ token, admin: { id: admin.id, username: admin.username } });
    } catch (error) {
        console.error('❌ Erro no login:', error);
        return res.status(500).json({ error: 'Erro ao fazer login' });
    }
});

app.post('/api/confirmar-pagamento', async (req, res) => {
    try {
        const { pagamento_id } = req.body;

        if (!pagamento_id) {
            return res.status(400).json({ error: 'pagamento_id obrigatório' });
        }

        const { data: pagamento, error: pagamentoError } = await supabase
            .from('pagamentos')
            .select('*')
            .eq('id', pagamento_id)
            .single();

        if (pagamentoError || !pagamento) {
            console.error('❌ Pagamento não encontrado:', pagamento_id);
            return res.status(404).json({ error: 'Pagamento não encontrado' });
        }

        if (pagamento.status === 'confirmed') {
            console.log(`⚠️ PAGAMENTO JÁ CONFIRMADO: ${pagamento_id}`);
            return res.json({ status: 'confirmado', mensagem: 'Pagamento já foi confirmado' });
        }

        const { error: updateError } = await supabase
            .from('pagamentos')
            .update({ status: 'confirmed' })
            .eq('id', pagamento_id);

        if (updateError) throw updateError;

        const { data: usuario } = await supabase
            .from('usuarios')
            .select('*')
            .eq('id', pagamento.usuario_id)
            .single();

        if (!usuario) {
            console.error('❌ Usuário não encontrado:', pagamento.usuario_id);
            return res.status(404).json({ error: 'Usuário não encontrado' });
        }

        const { data: plano } = await supabase
            .from('planos')
            .select('*')
            .eq('id', pagamento.plano_id)
            .single();

        if (!plano) {
            console.error('❌ Plano não encontrado:', pagamento.plano_id);
            return res.status(404).json({ error: 'Plano não encontrado' });
        }

        const { error: userError } = await supabase
            .from('usuarios')
            .update({
                plano_id: pagamento.plano_id,
                status: 'ativo',
                data_expiracao: new Date(new Date().getTime() + (plano.duracao_dias || 30) * 24 * 60 * 60 * 1000).toISOString()
            })
            .eq('id', pagamento.usuario_id);

        if (userError) throw userError;

        const dataExpiracao = new Date();
        dataExpiracao.setDate(dataExpiracao.getDate() + (plano.duracao_dias || 30));

        const { data: sessao, error: sessaoError } = await supabase
            .from('acessos_hotspot')
            .insert({
                usuario_id: pagamento.usuario_id,
                pop_id: 'qualquer_pop',
                mac_address: 'pendente',
                acesso_liberado: true,
                status: 'active',
                expira_em: dataExpiracao.toISOString(),
                criado_em: new Date().toISOString()
            })
            .select();

        if (sessaoError) throw sessaoError;

        console.log(`✅ PAGAMENTO CONFIRMADO: ${pagamento_id}`);
        console.log(`✅ USUÁRIO ATUALIZADO: ${usuario.username}`);
        console.log(`✅ SESSÃO CRIADA: ${sessao[0].id} - Expira: ${dataExpiracao.toISOString()}`);

        return res.json({
            status: 'confirmado',
            pagamento_id,
            usuario: usuario.username,
            plano: plano.nome,
            sessao_id: sessao[0].id,
            expira_em: dataExpiracao.toISOString(),
            mensagem: 'Pagamento confirmado! Acesso liberado automaticamente.'
        });

    } catch (error) {
        console.error('❌ Erro ao confirmar pagamento:', error);
        res.status(500).json({ error: 'Erro ao confirmar pagamento', details: error.message });
    }
});

app.post('/api/liberar-teste', async (req, res) => {
    const { mac } = req.body;
    if (!mac) return res.status(400).json({ error: 'MAC não informado' });
    
    try {
        // ==========================================
        // 1. LÓGICA DE LIMITE DE TESTES
        // ==========================================
        const { data: existingTeste } = await supabase
            .from('testes_gratis')
            .select('*')
            .eq('mac', mac)
            .single();
        
        const agora = new Date();
        const umaHora = 60 * 60 * 1000;
        
        if (existingTeste) {
            const ultimoTeste = new Date(existingTeste.ultimo_teste);
            if (agora - ultimoTeste < umaHora) {
                const minutosRestantes = Math.ceil((umaHora - (agora - ultimoTeste)) / 60000);
                return res.status(429).json({ 
                    error: `Aguarde ${minutosRestantes} minutos para outro teste` 
                });
            }
        }
        
        // Atualizar registro de teste
        await supabase
            .from('testes_gratis')
            .upsert({ 
                mac, 
                ultimo_teste: agora.toISOString(),
                tentativas: (existingTeste?.tentativas || 0) + 1
            });
        
        // ==========================================
        // 2. LIBERAR ACESSO (FUNÇÃO DUAL)
        // ==========================================
        const resultado = await liberarAcessoDual(mac);
        
        if (resultado.success) {
            console.log(`✅ Teste grátis liberado para MAC: ${mac} (API: ${resultado.viaAPI}, RADIUS: ${resultado.viaRADIUS})`);
            res.json({ 
                success: true, 
                message: 'Teste grátis liberado por 15 minutos',
                metodo: resultado.viaAPI ? 'api_local' : 'radius'
            });
        } else {
            console.error(`❌ Falha ao liberar MAC ${mac}: ${resultado.erros.join(', ')}`);
            res.status(500).json({ 
                error: 'Falha ao liberar acesso. Tente novamente.',
                details: resultado.erros
            });
        }
    } catch (error) {
        console.error('❌ Erro ao liberar teste:', error);
        res.status(500).json({ error: error.message });
    }
});

    try {
        const { username, mac_address, pop_id } = req.body;

            return res.json({ acesso: false, motivo: 'Parâmetros faltando' });
        }

        const { data: usuario } = await supabase
            .from('usuarios')
            .select('*')
            .eq('username', username)
            .single();

        if (!usuario) {
            return res.json({ acesso: false, motivo: 'Usuário não encontrado' });
        }

        const { data: sessao } = await supabase
            .from('acessos_hotspot')
            .select('*')
            .eq('usuario_id', usuario.id)
            .eq('status', 'active')
            .gte('expira_em', new Date().toISOString())
            .single();

        if (!sessao) {
            return res.json({ acesso: false, motivo: 'Sem sessão ativa' });
        }

        await supabase
            .from('acessos_hotspot')
            .update({
                mac_address,
                pop_id,
                atualizado_em: new Date().toISOString()
            })
            .eq('id', sessao.id);

        console.log(`✅ ACESSO VALIDADO: ${username} (MAC: ${mac_address})`);

        return res.json({
            acesso: true,
            usuario: usuario.username,
            sessao_id: sessao.id,
            expira_em: sessao.expira_em,
            motivo: 'Acesso ativo'
        });

    } catch (error) {
        console.error('❌ Erro ao validar acesso:', error);
        return res.json({ acesso: false, motivo: 'Erro ao validar' });
    }
});

app.get('/api/planos/:id/detalhes', async (req, res) => {
    try {
        const { id } = req.params;
        
        const { data: plano, error: erroPlano } = await supabase
            .from('planos')
            .select('*')
            .eq('id', id)
            .single();
        
        if (erroPlano) throw erroPlano;
        
        const { data: usuarios, error: erroUsuarios } = await supabase
            .from('radreply')
            .select('id')
            .eq('plano', plano.nome)
            .eq('status', 'ativo');
        
        if (erroUsuarios) throw erroUsuarios;
        
        const faturamento = (usuarios?.length || 0) * plano.preco;
        
        res.json({ 
            success: true, 
            detalhes: {
                ...plano,
                assinantes_ativos: usuarios?.length || 0,
                faturamento_mensal: faturamento
            }
        });
    } catch (error) {
        console.error('❌ Erro ao buscar detalhes do plano:', error);
        res.status(500).json({ error: error.message });
    }
});

app.post('/api/pops/create', async (req, res) => {
    try {
        const {
            nome_pop,
            tipo_interface_clientes,
            interface_clientes,
            vlan_id,
            criar_bridge,
            nome_bridge,
            interface_wan,
            rede_hotspot,
            pool_dhcp_inicio,
            pool_dhcp_fim,
            dns_servers,
            hotspot_server_name,
            usar_radius,
            manter_config_existente
        } = req.body;

        if (!nome_pop || !interface_clientes) {
            return res.status(400).json({ error: 'nome_pop e interface_clientes são obrigatórios' });
        }

        const pop_id = 'MS-' + Math.random().toString(36).substr(2, 5).toUpperCase();

        const { data, error } = await supabase.from('pops').insert([{
            id: pop_id,
            nome: nome_pop,
            ip_publico: 'AUTO',
            status: 'offline',
            ultima_conexao: new Date().toISOString(),
            banda_utilizada: '0 Mbps',
            usuarios_conectados: 0,
            tipo_interface_clientes,
            interface_clientes,
            vlan_id: vlan_id || null,
            criar_bridge: criar_bridge || false,
            nome_bridge: nome_bridge || null,
            interface_wan: interface_wan || 'ether1',
            rede_hotspot: rede_hotspot || '192.168.32.1/20',
            pool_dhcp_inicio: pool_dhcp_inicio || '192.168.32.2',
            pool_dhcp_fim: pool_dhcp_fim || '192.168.47.254',
            dns_servers: dns_servers || '8.8.8.8,8.8.4.4',
            hotspot_server_name: hotspot_server_name || 'MS-' + pop_id,
            usar_radius: usar_radius || false,
            manter_config_existente: manter_config_existente || true
        }]).select();

        if (error) throw error;

        console.log(`✅ POP criado: ${pop_id}`);
        return res.json({ pop_id, pop: data[0], message: 'POP criado com sucesso' });

    } catch (error) {
        console.error('❌ Erro ao criar POP:', error);
        res.status(500).json({ error: 'Erro ao criar POP', details: error.message });
    }
});

    try {
        const { mac } = req.query;

        if (!mac) {
            return res.status(400).json({ error: 'MAC obrigatório' });
        }

        const { data: pagamento, error } = await supabase
            .from('pagamentos')
            .select('*')
            .eq('usuario_mac', mac)
            .eq('status', 'confirmed')
            .order('criado_em', { ascending: false })
            .limit(1)
            .single();

        if (error || !pagamento) {
            return res.json({ pago: false });
        }

        return res.json({
            pago: true,
            pagamento_id: pagamento.id,
            valor: pagamento.valor,
            data: pagamento.criado_em
        });

    } catch (error) {
        console.error('❌ Erro ao verificar pagamento:', error);
        res.status(500).json({ error: error.message });
    }
});

function gerarScriptInstalacao(params) {
    const {
        UNIQUE_ID, POP_NAME,
        TIPO_INSTALACAO,
        WAN_INTERFACE, LAN_INTERFACE, VLAN_ID,
        WAN_TIPO, PPPOE_USER, PPPOE_PASS, STATIC_IP, STATIC_MASK, STATIC_GW,
        HOTSPOT_TIPO, VLAN_EXISTENTE, PORTA_FISICA,
        RADIUS_SECRET
    } = params;

    const timestamp = new Date().toISOString();
    const scheduler_name = removerAcentos(POP_NAME).replace(/[^a-zA-Z0-9]/g, '-');

    // ==================== CONFIGURAÇÃO DA SAÍDA (LAN) ====================
    let configSaida = '';
    let interfaceSaida = LAN_INTERFACE;
    let configBridgePort = '';

    if (HOTSPOT_TIPO === 'vlan_existente' && VLAN_EXISTENTE) {
        configSaida = `
# 4. VLAN EXISTENTE
/interface vlan add name="vlan-${UNIQUE_ID}" vlan-id=${VLAN_EXISTENTE} interface=${LAN_INTERFACE} comment="MS-TELECOM-${UNIQUE_ID}"
:delay 500ms`;
        interfaceSaida = `vlan-${UNIQUE_ID}`;
        configBridgePort = `/interface bridge port add bridge="ms-bridge-${UNIQUE_ID}" interface="${interfaceSaida}" comment="MS-TELECOM-${UNIQUE_ID}"\n:delay 1000ms`;
    } else if (HOTSPOT_TIPO === 'porta_fisica' && PORTA_FISICA) {
        configSaida = `
# 4. PORTA FÍSICA DEDICADA
:delay 500ms`;
        interfaceSaida = PORTA_FISICA;
        configBridgePort = `/interface bridge port add bridge="ms-bridge-${UNIQUE_ID}" interface="${interfaceSaida}" comment="MS-TELECOM-${UNIQUE_ID}"\n:delay 1000ms`;
    } else if (VLAN_ID && VLAN_ID !== '') {
        configSaida = `
# 4. VLAN
/interface vlan add name="vlan-hotspot-${UNIQUE_ID}" vlan-id=${VLAN_ID} interface=${LAN_INTERFACE} comment="MS-TELECOM-${UNIQUE_ID}"`;
        interfaceSaida = `vlan-hotspot-${UNIQUE_ID}`;
        configBridgePort = `/interface bridge port add bridge="ms-bridge-${UNIQUE_ID}" interface="${interfaceSaida}" comment="MS-TELECOM-${UNIQUE_ID}"\n:delay 1000ms`;
    } else {
        configBridgePort = `/interface bridge port add bridge="ms-bridge-${UNIQUE_ID}" interface="${interfaceSaida}" comment="MS-TELECOM-${UNIQUE_ID}"\n:delay 1000ms`;
    }

    // ==================== CONFIGURAÇÃO DA WAN ====================
    let configWan = '';

    if (TIPO_INSTALACAO === 'novo') {
        if (WAN_TIPO === 'dhcp') {
            configWan = `
# 7. DHCP CLIENT (WAN)
/ip dhcp-client add interface=${WAN_INTERFACE} disabled=no comment="MS-TELECOM-${UNIQUE_ID}"
:delay 500ms`;
        } else if (WAN_TIPO === 'pppoe') {
            configWan = `
# 7. PPPoE CLIENT (WAN)
/interface pppoe-client add interface=${WAN_INTERFACE} user="${PPPOE_USER}" password="${PPPOE_PASS}" disabled=no comment="MS-TELECOM-${UNIQUE_ID}" add-default-route=yes
:delay 500ms`;
        } else if (WAN_TIPO === 'static') {
            configWan = `
# 7. IP ESTÁTICO (WAN)
/ip address add address=${STATIC_IP}/${STATIC_MASK} interface=${WAN_INTERFACE} comment="MS-TELECOM-${UNIQUE_ID}"
/ip route add gateway=${STATIC_GW} comment="MS-TELECOM-${UNIQUE_ID}"
:delay 500ms`;
        } else {
            configWan = `
# 7. DHCP CLIENT (WAN) - PADRÃO
/ip dhcp-client add interface=${WAN_INTERFACE} disabled=no comment="MS-TELECOM-${UNIQUE_ID}"
:delay 500ms`;
        }
    } else {
        configWan = `
# WAN já configurada (modo produção) – pulando configuração
:delay 500ms`;
    }

    // ==================== CONSTRUÇÃO DO SCRIPT ====================
    return `# ============================================
# MS TELECOM - SCRIPT DE INSTALAÇÃO
# ID: ${UNIQUE_ID}
# POP: ${POP_NAME}
# DATA: ${timestamp}
# TIPO: ${TIPO_INSTALACAO === 'novo' ? 'Novo/Resetado' : 'Em Produção'}
# WAN: ${WAN_TIPO || 'Não configurada'}
# ============================================

# 1. BACKUP OBRIGATÓRIO
/system backup save name=backup_pre_ms_${UNIQUE_ID}
/export file=config_pre_ms_${UNIQUE_ID}
:delay 2000ms

# 2. IDENTIDADE
/system identity set name="${POP_NAME}"
:delay 500ms

# 3. BRIDGE
/interface bridge add name="ms-bridge-${UNIQUE_ID}" comment="MS-TELECOM-${UNIQUE_ID}"
:delay 500ms

${configSaida}
${configBridgePort}

# 5. IP DO HOTSPOT
/ip address add address=192.168.32.1/20 interface="ms-bridge-${UNIQUE_ID}" network=192.168.32.0 comment="MS-TELECOM-${UNIQUE_ID}"
:delay 500ms

# 6. DHCP (LAN)
/ip pool add name="ms-pool-${UNIQUE_ID}" ranges=192.168.32.10-192.168.47.254 comment="MS-TELECOM-${UNIQUE_ID}"
:delay 500ms
/ip dhcp-server add address-pool="ms-pool-${UNIQUE_ID}" disabled=no interface="ms-bridge-${UNIQUE_ID}" name="ms-dhcp-${UNIQUE_ID}" lease-time=24h
:delay 500ms
/ip dhcp-server network add address=192.168.32.0/20 gateway=192.168.32.1 dns-server=8.8.8.8,1.1.1.1 comment="MS-TELECOM-${UNIQUE_ID}"
:delay 500ms

${configWan}

# 8. NAT (WAN)
/ip firewall nat add action=masquerade chain=srcnat out-interface=${WAN_INTERFACE} comment="MS-TELECOM-${UNIQUE_ID}"
:delay 500ms

# 9. DNS
/ip dns set allow-remote-requests=yes servers=8.8.8.8,1.1.1.1
:delay 500ms

# 10. RADIUS
/radius add address=40.233.118.238 secret=${RADIUS_SECRET} service=hotspot comment="MS-TELECOM-${UNIQUE_ID}" timeout=1000ms domain="${UNIQUE_ID}"
/radius incoming set accept=yes
:delay 1000ms

# 11. HOTSPOT PROFILE
:global hotspotDir
:set hotspotDir "ms-${UNIQUE_ID}"
/ip hotspot profile add name="ms-profile-${UNIQUE_ID}" hotspot-address=192.168.32.1 html-directory=$hotspotDir login-by=http-chap,http-pap use-radius=yes radius-default-domain="${UNIQUE_ID}" radius-interim-update="10m"
:delay 500ms

# 12. HOTSPOT
/ip hotspot add address-pool="ms-pool-${UNIQUE_ID}" disabled=no idle-timeout=15m interface="ms-bridge-${UNIQUE_ID}" name="${POP_NAME}" profile="ms-profile-${UNIQUE_ID}"
:delay 1000ms

# 13. WALLED-GARDEN
/ip hotspot walled-garden ip add action=accept disabled=no dst-host=hotspot-system.vercel.app server="${POP_NAME}" comment="MS-TELECOM-${UNIQUE_ID}"
/ip hotspot walled-garden ip add action=accept disabled=no dst-host=cdn.tailwindcss.com server="${POP_NAME}"
/ip hotspot walled-garden ip add action=accept disabled=no dst-host=unpkg.com server="${POP_NAME}"

/ip hotspot walled-garden ip add action=accept disabled=no dst-host=fonts.gstatic.com server="${POP_NAME}"
:delay 500ms

# 14. HEARTBEAT (SCHEDULER)
/system scheduler add name="heartbeat-${scheduler_name}" interval=30s on-event="/tool fetch http-method=post url=\\"http://mstelecom-api.duckdns.org/api/ping\\" http-data=\\"{\\\\\\"pop_id\\\\\\":\\\\\\"${POP_NAME}\\\\\\"}\\" keep-result=no" start-time=startup
:delay 1000ms

:execute {/ip hotspot reset-html "${POP_NAME}"}
:delay 5000ms

:global hotspotDir [/ip hotspot profile get [find name="ms-profile-${UNIQUE_ID}"] value-name=html-directory]

# Criar login.html com JavaScript
/file set [:put (\$hotspotDir."/login.html")] contents="<html><head><title>MS Telecom</title><script>var hotspotIdentity='\$(server-name)';var userMac='\$(mac)';var hostname='\$(ip)';var error='\$(error)';window.location.href='http://mstelecom-api.duckdns.org/entrypoint?hotspotIdentity=' + encodeURIComponent(hotspotIdentity) + '&userMac=' + encodeURIComponent(userMac) + '&hostname=' + encodeURIComponent(hostname) + '&error=' + encodeURIComponent(error);</script></head><body style='background:#0a0c15;color:white;font-family:sans-serif;display:flex;align-items:center;justify-content:center;height:100vh'><div style='text-align:center'><div style='width:50px;height:50px;border:3px solid #1e293b;border-top-color:#3b82f6;border-radius:50%;animation:spin 1s linear infinite;margin:0 auto 20px'></div><h2>MS TELECOM</h2><p>Conectando...</p></div><style>@keyframes spin{to{transform:rotate(360deg)}}</style></body></html>"

app.post('/api/webhooks/:id/testar', async (req, res) => {
    try {
        const { id } = req.params;

        const { data: webhook, error: webhookError } = await supabase
            .from('webhooks')
            .select('*')
            .eq('id', id)
            .single();

        if (webhookError || !webhook) {
            return res.status(404).json({ error: 'Webhook não encontrado' });
        }

        const testPayload = {
            test: true,
            message: 'Teste de webhook',
            timestamp: new Date().toISOString(),
            evento: webhook.evento
        };

        try {
            const response = await fetch(webhook.url, {
                method: webhook.metodo || 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(testPayload),
                timeout: 5000
            });

            if (response.ok) {
                await supabase
                    .from('webhooks')
                    .update({
                        ultima_execucao: new Date().toISOString(),
                        total_eventos: (webhook.total_eventos || 0) + 1
                    })
                    .eq('id', id);

                console.log(`✅ Webhook testado: ${id}`);
                res.json({ success: true, message: 'Teste enviado com sucesso' });
            } else {
                res.status(500).json({ error: `Falha ao enviar teste: ${response.status}` });
            }
        } catch (fetchError) {
            res.status(500).json({ error: `Erro ao conectar: ${fetchError.message}` });
        }
    } catch (error) {
        console.error('❌ Erro ao testar webhook:', error);
        res.status(500).json({ error: error.message });
    }
});

// ===============================
app.post('/api/liberar-teste', async (req, res) => {
    try {
        const { mac } = req.body;

        if (!mac) {
            return res.status(400).json({ error: 'MAC obrigatório' });
        }

        const { data: testeExistente } = await supabase
            .from('testes_gratis')
            .select('*')
            .eq('mac', mac)
            .single();

        const agora = new Date();
        const umaHora = 60 * 60 * 1000;

        if (testeExistente) {
            const ultimoTeste = new Date(testeExistente.ultimo_teste);
            const diferenca = agora - ultimoTeste;

            if (diferenca < umaHora) {
                const minutosRestantes = Math.ceil((umaHora - diferenca) / 60000);
                console.log(`⚠️ Teste grátis já usado. Aguarde ${minutosRestantes} minutos`);
                return res.status(429).json({
                    error: `Você já usou o teste grátis. Aguarde ${minutosRestantes} minutos para tentar novamente`
                });
            }

            await supabase
                .from('testes_gratis')
                .update({
                    ultimo_teste: agora.toISOString(),
                    tentativas: testeExistente.tentativas + 1
                })
                .eq('mac', mac);
        } else {
            await supabase
                .from('testes_gratis')
                .insert([{ mac, ultimo_teste: agora.toISOString(), tentativas: 1 }]);
        }

        console.log(`✅ Teste grátis liberado para MAC: ${mac}`);
        res.json({ success: true, message: '5 minutos liberados', expira_em: 5 * 60 });
    } catch (error) {
        console.error('❌ Erro ao liberar teste:', error);
        res.status(500).json({ error: error.message });
    }
});

app.get('/api/usuarios/export', async (req, res) => {
    try {
        const { data, error } = await supabase
            .from('usuarios')
            .select('*')
            .order('created_at', { ascending: false });

        if (error) throw error;

        const headers = ['Nome', 'Telefone', 'CPF', 'Email', 'Plano', 'Status', 'Data Cadastro'];
        const rows = (data || []).map(u => [
            u.nome || '',
            u.telefone || '',
            u.cpf || '',
            u.email || '',
            u.plano || '',
            u.status || 'ativo',
            u.created_at ? new Date(u.created_at).toLocaleDateString('pt-BR') : ''
        ]);

        const csv = [headers, ...rows].map(row => row.map(cell => `"${cell}"`).join(';')).join('\n');

        res.setHeader('Content-Type', 'text/csv; charset=utf-8');
        res.setHeader('Content-Disposition', `attachment; filename=clientes_${new Date().toISOString().slice(0,10)}.csv`);
        res.send('\uFEFF' + csv);

        console.log(`✅ Dados exportados: ${rows.length} registros`);
    } catch (error) {
        console.error('❌ Erro ao exportar dados:', error);
        res.status(500).json({ error: error.message });
    }
});

// ===============================
app.get('/api/usuarios/filtro', async (req, res) => {
    try {
        const { nome, status, plano, data_inicio, data_fim } = req.query;

        let query = supabase.from('usuarios').select('*');

        if (nome) {
            query = query.ilike('nome', `%${nome}%`);
        }
        if (status) {
            query = query.eq('status', status);
        }
        if (plano) {
            query = query.eq('plano', plano);
        }
        if (data_inicio) {
            query = query.gte('created_at', new Date(data_inicio).toISOString());
        }
        if (data_fim) {
            query = query.lte('created_at', new Date(data_fim).toISOString());
        }

        const { data, error } = await query.order('created_at', { ascending: false });

        if (error) throw error;
        console.log(`✅ Filtros aplicados: ${data?.length || 0} resultados`);
        res.json(data || []);
    } catch (error) {
        console.error('❌ Erro ao filtrar usuários:', error);
        res.status(500).json({ error: error.message });
    }
});
